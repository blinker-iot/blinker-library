cmake_minimum_required(VERSION 3.13)
project(blinker_wio_rpc_pending_tests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(WIO_RPC_DIR "" CACHE PATH "Read-only pinned rpcUnified 2.1.4 library root")
set(WIO_SDK_ROOT "" CACHE PATH "Original SDK path fence, not an SDK verification")
if(NOT EXISTS "${WIO_RPC_DIR}/src/erpc/erpc_client_manager.cpp" OR NOT IS_DIRECTORY "${WIO_SDK_ROOT}")
    message(FATAL_ERROR "Explicit original RPC library and SDK path fence required")
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(OVERLAY "${CMAKE_BINARY_DIR}/overlay")
set(verify_flags "")
if(EXISTS "${OVERLAY}")
    set(verify_flags --verify-only)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${REPO_ROOT}/tools/prepare_wio_rpc_overlay.py"
    --rpc-dir "${WIO_RPC_DIR}" --sdk-root "${WIO_SDK_ROOT}" --output "${OVERLAY}" ${verify_flags}
    RESULT_VARIABLE overlay_result OUTPUT_VARIABLE overlay_stdout ERROR_VARIABLE overlay_stderr)
file(WRITE "${CMAKE_BINARY_DIR}/overlay-generation.log" "${overlay_stdout}\n${overlay_stderr}")
if(NOT overlay_result EQUAL 0)
    message(FATAL_ERROR "Pinned overlay generation/verification failed; preserve overlay-generation.log")
endif()

# Real SDK codec/manager implementations and headers, copied unmodified. Only
# OS/Arduino primitives are host seams. The pending algorithm comes exclusively
# from the generated overlay, never a fixture reimplementation.
set(PROJECTION "${CMAKE_BINARY_DIR}/projection")
file(MAKE_DIRECTORY "${PROJECTION}")
file(GLOB sdk_headers "${WIO_RPC_DIR}/src/erpc/*.h")
foreach(file IN LISTS sdk_headers)
    get_filename_component(name "${file}" NAME)
    configure_file("${file}" "${PROJECTION}/${name}" COPYONLY)
endforeach()
set(original_units erpc_client_manager.cpp erpc_basic_codec.cpp erpc_message_buffer.cpp erpc_message_loggers.cpp)
set(units "")
foreach(name IN LISTS original_units)
    configure_file("${WIO_RPC_DIR}/src/erpc/${name}" "${PROJECTION}/${name}" COPYONLY)
    list(APPEND units "${PROJECTION}/${name}")
endforeach()
foreach(name IN ITEMS erpc_transport_arbitrator.h erpc_transport_arbitrator.cpp erpc_arbitrated_client_manager.cpp)
    configure_file("${OVERLAY}/erpc/${name}" "${PROJECTION}/${name}" COPYONLY)
    if(name MATCHES "\\.cpp$")
        list(APPEND units "${PROJECTION}/${name}")
    endif()
endforeach()
configure_file(stubs/erpc_threading.h "${PROJECTION}/erpc_threading.h" COPYONLY)
configure_file(stubs/Arduino.h "${PROJECTION}/Arduino.h" COPYONLY)

foreach(mode IN ITEMS native logging)
    add_executable(wio_rpc_pending_${mode} fixture.cpp ${units})
    target_include_directories(wio_rpc_pending_${mode} PRIVATE "${PROJECTION}")
    target_compile_definitions(wio_rpc_pending_${mode} PRIVATE SEEED_WIO_TERMINAL=1
        BLINKER_ENABLE_TASKS=1 BLINKER_INTERNAL_WIO_TASKS=1
        BLINKER_INTERNAL_WIO_TASK_REGION_RESERVED=1 BLINKER_TASK_CAPACITY=4)
    if(mode STREQUAL "logging")
        target_compile_definitions(wio_rpc_pending_${mode} PRIVATE ERPC_MESSAGE_LOGGING=1)
    endif()
    if(MSVC)
        target_compile_options(wio_rpc_pending_${mode} PRIVATE /W3)
    else()
        target_compile_options(wio_rpc_pending_${mode} PRIVATE -Wall -Wextra)
        target_link_libraries(wio_rpc_pending_${mode} PRIVATE pthread)
    endif()
endforeach()
foreach(case IN ITEMS atomic-publish early-reply duplicate-reply send-failure-late
        stale-signal free-list-next wait-false receive-before-cancel cancel-during-delivery excessive-signals)
    add_test(NAME wio_rpc_${case} COMMAND wio_rpc_pending_native ${case})
    set_tests_properties(wio_rpc_${case} PROPERTIES TIMEOUT 15)
endforeach()
add_test(NAME wio_rpc_logging_failure COMMAND wio_rpc_pending_logging logging-failure)
set_tests_properties(wio_rpc_logging_failure PROPERTIES TIMEOUT 15)
