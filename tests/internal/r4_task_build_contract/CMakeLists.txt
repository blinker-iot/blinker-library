cmake_minimum_required(VERSION 3.13)
project(blinker_internal_r4_task_build_contract LANGUAGES CXX)
enable_testing()

function(fixture target)
    add_executable(${target} EXCLUDE_FROM_ALL test_main.cpp providers.cpp)
    target_compile_features(${target} PRIVATE cxx_std_11)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src")
endfunction()

fixture(r4_task_build_contract)
set_target_properties(r4_task_build_contract PROPERTIES EXCLUDE_FROM_ALL FALSE)
foreach(case normal rpc-fault heap-unavailable)
    add_test(NAME r4_task_contract_${case} COMMAND r4_task_build_contract "${case}")
endforeach()

fixture(r4_task_build_disabled)
target_compile_definitions(r4_task_build_disabled PRIVATE FIXTURE_DISABLED)
set_target_properties(r4_task_build_disabled PROPERTIES EXCLUDE_FROM_ALL FALSE)
add_test(NAME r4_task_contract_disabled COMMAND r4_task_build_disabled)

function(reject name definition expected)
    set(target "r4_task_reject_${name}")
    fixture(${target})
    target_compile_definitions(${target} PRIVATE "${definition}")
    add_test(NAME ${target} COMMAND "${CMAKE_COMMAND}"
        "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DCONFIG=$<CONFIG>" "-DTARGET=${target}"
        "-DEXPECT=${expected}" -P "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.cmake")
endfunction()
