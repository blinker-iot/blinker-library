cmake_minimum_required(VERSION 3.13)
project(blinker_internal_r4_modem_uart LANGUAGES CXX)
enable_testing()
if(NOT DEFINED R4_UART_OVERLAY OR NOT EXISTS "${R4_UART_OVERLAY}/Serial.cpp")
    message(FATAL_ERROR "Supply a hash-bound R4_UART_OVERLAY with Serial.cpp/Serial.h")
endif()
if(NOT DEFINED R4_UART_CORE_SOURCE)
    message(FATAL_ERROR "Supply the untouched SDK cores/arduino source")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(uart_verifier "${CMAKE_CURRENT_SOURCE_DIR}/../../../tools/r4_modem_uart_overlay.py")
set(uart_check "${Python3_EXECUTABLE}" "${uart_verifier}" --core-dir "${R4_UART_CORE_SOURCE}"
    --output "${R4_UART_OVERLAY}" --verify-only)
execute_process(COMMAND ${uart_check} RESULT_VARIABLE uart_result)
if(NOT uart_result EQUAL 0)
    message(FATAL_ERROR "Prepared UART overlay byte verification failed")
endif()
add_executable(r4_modem_uart_tests test_main.cpp "${R4_UART_OVERLAY}/Serial.cpp")
add_custom_target(r4_modem_uart_overlay_verify COMMAND ${uart_check} VERBATIM)
add_dependencies(r4_modem_uart_tests r4_modem_uart_overlay_verify)
target_compile_features(r4_modem_uart_tests PRIVATE cxx_std_11)
target_include_directories(r4_modem_uart_tests PRIVATE stubs "${R4_UART_OVERLAY}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../src")
if(MSVC)
    target_compile_options(r4_modem_uart_tests PRIVATE /W4)
else()
    target_compile_options(r4_modem_uart_tests PRIVATE -Wall -Wextra)
endif()
foreach(case normal byte-timeout bulk-timeout write-error flush-timeout expired-before-tx
        late-callback close-failure transfer-refused other-uart const-print raw-timeout inactive
        fault-during-tx fault-during-flush const-whole-budget)
    add_test(NAME r4_uart_${case} COMMAND r4_modem_uart_tests "${case}")
endforeach()
